% Бүртгэл ба нэвтрэлтийн дарааллын диаграмм — standalone TikZ
% Compile: xelatex seq_auth.tex
\documentclass[border=10pt]{standalone}
\usepackage{fontspec}
\setmainfont{Times New Roman}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, calc}

\begin{document}

\begin{tikzpicture}[
  % ---- styles ----
  participant/.style = {
    rectangle, rounded corners=3pt,
    minimum width=2.8cm, minimum height=0.9cm,
    draw=black!70, line width=0.6pt,
    fill=blue!8,
    align=center, font=\small\bfseries
  },
  msg/.style = {-Stealth, thick, black!70},
  ret/.style = {-Stealth, dashed, thick, black!50},
  msglabel/.style = {font=\scriptsize, fill=white, inner sep=2pt},
  note/.style = {
    rectangle, rounded corners=2pt,
    draw=black!40, fill=yellow!10,
    font=\scriptsize, align=left, inner sep=4pt
  }
]

% ---- Participants ----
\node[participant] (user)     at (0, 0)    {Хэрэглэгч\\(Хөтөч)};
\node[participant] (nextauth) at (4.5, 0)  {NextAuth.js};
\node[participant] (prisma)   at (9, 0)    {Prisma ORM};
\node[participant] (db)       at (13.5, 0) {PostgreSQL};

% ---- Lifelines ----
\draw[dashed, black!30] (user.south)     -- ++(0, -14.5);
\draw[dashed, black!30] (nextauth.south) -- ++(0, -14.5);
\draw[dashed, black!30] (prisma.south)   -- ++(0, -14.5);
\draw[dashed, black!30] (db.south)       -- ++(0, -14.5);

% ---- Activation boxes ----
\fill[blue!6, draw=black!40, line width=0.4pt]
  ($(user.south)     + (-0.15, -0.3)$)  rectangle ++(0.3, -13.8);
\fill[blue!6, draw=black!40, line width=0.4pt]
  ($(nextauth.south) + (-0.15, -1.2)$)  rectangle ++(0.3, -11.5);
\fill[blue!6, draw=black!40, line width=0.4pt]
  ($(prisma.south)   + (-0.15, -3.0)$)  rectangle ++(0.3, -5.5);
\fill[blue!6, draw=black!40, line width=0.4pt]
  ($(db.south)       + (-0.15, -4.0)$)  rectangle ++(0.3, -3.0);

% ============ БҮРТГЭЛ (Registration) ============

% Group box
\draw[black!30, line width=0.5pt, rounded corners=3pt]
  (-1.5, -0.8) rectangle (15, -7.5);
\node[font=\scriptsize\bfseries, fill=white, inner sep=3pt] at (-0.2, -0.8) {Бүртгэл};

% 1. Хэрэглэгч → NextAuth: Бүртгэлийн мэдээлэл илгээх
\draw[msg] ($(user.south) + (0.15, -1.3)$)
  -- node[msglabel, above] {1: Бүртгэлийн мэдээлэл (нэр, имэйл, нууц үг)}
  ($(nextauth.south) + (-0.15, -1.3)$);

% 2. NextAuth → Prisma: Хэрэглэгч үүсгэх
\draw[msg] ($(nextauth.south) + (0.15, -2.3)$)
  -- node[msglabel, above] {2: Шинэ хэрэглэгч үүсгэх хүсэлт}
  ($(prisma.south) + (-0.15, -2.3)$);

% 3. Prisma → DB: INSERT
\draw[msg] ($(prisma.south) + (0.15, -3.3)$)
  -- node[msglabel, above] {3: INSERT INTO users}
  ($(db.south) + (-0.15, -3.3)$);

% 4. DB → Prisma: OK
\draw[ret] ($(db.south) + (-0.15, -4.5)$)
  -- node[msglabel, above] {4: Хэрэглэгч амжилттай үүссэн}
  ($(prisma.south) + (0.15, -4.5)$);

% 5. Prisma → NextAuth: Created user
\draw[ret] ($(prisma.south) + (-0.15, -5.5)$)
  -- node[msglabel, above] {5: User объект буцаах}
  ($(nextauth.south) + (0.15, -5.5)$);

% 6. NextAuth → User: Бүртгэл амжилттай
\draw[ret] ($(nextauth.south) + (-0.15, -6.5)$)
  -- node[msglabel, above] {6: Бүртгэл амжилттай, нэвтрэх хуудас руу}
  ($(user.south) + (0.15, -6.5)$);

% ============ НЭВТРЭЛТ (Login) ============

% Group box
\draw[black!30, line width=0.5pt, rounded corners=3pt]
  (-1.5, -7.9) rectangle (15, -14.6);
\node[font=\scriptsize\bfseries, fill=white, inner sep=3pt] at (-0.1, -7.9) {Нэвтрэлт};

% 7. User → NextAuth: Имэйл/Нууц үг
\draw[msg] ($(user.south) + (0.15, -8.3)$)
  -- node[msglabel, above] {7: Имэйл, нууц үг оруулах}
  ($(nextauth.south) + (-0.15, -8.3)$);

% 8. NextAuth → Prisma: Find user
\draw[msg] ($(nextauth.south) + (0.15, -9.3)$)
  -- node[msglabel, above] {8: Имэйлээр хэрэглэгч хайх}
  ($(prisma.south) + (-0.15, -9.3)$);

% 9. Prisma → DB: SELECT
\draw[msg] ($(prisma.south) + (0.15, -10.0)$)
  -- node[msglabel, above] {9: SELECT * FROM users WHERE email=?}
  ($(db.south) + (-0.15, -10.0)$);

% 10. DB → Prisma: User data
\draw[ret] ($(db.south) + (-0.15, -10.9)$)
  -- node[msglabel, above] {10: Хэрэглэгчийн мэдээлэл}
  ($(prisma.south) + (0.15, -10.9)$);

% 11. Prisma → NextAuth: bcrypt шалгах
\draw[ret] ($(prisma.south) + (-0.15, -11.8)$)
  -- node[msglabel, above] {11: User + hashed password}
  ($(nextauth.south) + (0.15, -11.8)$);

% 12. NextAuth → User: JWT токен
\draw[ret] ($(nextauth.south) + (-0.15, -13.0)$)
  -- node[msglabel, above] {12: JWT session токен (cookie-д хадгалах)}
  ($(user.south) + (0.15, -13.0)$);

% Note
\node[note] at ($(nextauth.south) + (2.5, -12.5)$) {bcrypt.compare()\\нууц үг шалгах};

% Bottom participants (repeat)
\node[participant] (user2)     at (0, -15.2)    {Хэрэглэгч\\(Хөтөч)};
\node[participant] (nextauth2) at (4.5, -15.2)  {NextAuth.js};
\node[participant] (prisma2)   at (9, -15.2)    {Prisma ORM};
\node[participant] (db2)       at (13.5, -15.2) {PostgreSQL};

\end{tikzpicture}
\end{document}
